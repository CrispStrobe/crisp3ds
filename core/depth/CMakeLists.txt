# This directory is experimental and is added only by the OpenCV test build.
add_library(crisp3ds_depth_geometry STATIC depth_geometry.cpp)
target_include_directories(crisp3ds_depth_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CRISP3DS_OPENCV_INCLUDE_DIRS})
target_link_libraries(crisp3ds_depth_geometry PUBLIC ${CRISP3DS_OPENCV_LIBRARIES})
target_compile_features(crisp3ds_depth_geometry PUBLIC cxx_std_20)

add_executable(crisp3ds_depth_geometry_tests tests.cpp)
target_link_libraries(crisp3ds_depth_geometry_tests PRIVATE crisp3ds_depth_geometry)
add_test(NAME depth_geometry_synthetic COMMAND crisp3ds_depth_geometry_tests)

add_executable(crisp3ds_depth_image_tests image_tests.cpp)
target_link_libraries(crisp3ds_depth_image_tests PRIVATE crisp3ds_depth_geometry)
add_test(NAME depth_image_synthetic COMMAND crisp3ds_depth_image_tests ${CMAKE_BINARY_DIR}/depth-image-fixtures)

add_executable(crisp3ds_depth_image_fusion_tests image_fusion_tests.cpp)
target_link_libraries(crisp3ds_depth_image_fusion_tests PRIVATE crisp3ds_depth_geometry)
add_test(NAME depth_image_fusion_synthetic COMMAND crisp3ds_depth_image_fusion_tests ${CMAKE_BINARY_DIR}/depth-image-fusion-fixtures)
