cmake_minimum_required(VERSION 3.20)
project(crisp3ds VERSION 0.1.0 LANGUAGES CXX)
include(cmake/OpenCV.cmake)

option(CRISP3DS_BUILD_TESTS "Build native tests" ON)

add_library(crisp3ds_core STATIC core/src/core.cpp)
target_include_directories(crisp3ds_core PUBLIC core/include)
target_compile_features(crisp3ds_core PUBLIC cxx_std_20)
set_target_properties(crisp3ds_core PROPERTIES CXX_EXTENSIONS OFF)
if(CRISP3DS_WITH_OPENCV)
  target_link_libraries(crisp3ds_core PUBLIC ${CRISP3DS_OPENCV_LIBRARIES})
  target_include_directories(crisp3ds_core PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
  target_compile_definitions(crisp3ds_core PUBLIC CRISP3DS_WITH_OPENCV=1)
endif()

add_executable(crisp3ds core/src/cli.cpp)
target_link_libraries(crisp3ds PRIVATE crisp3ds_core)

if(CRISP3DS_BUILD_TESTS)
  enable_testing()
  add_executable(crisp3ds_tests core/tests/core_tests.cpp)
  target_link_libraries(crisp3ds_tests PRIVATE crisp3ds_core)
  add_test(NAME core_contract COMMAND crisp3ds_tests)
  add_test(NAME cli_capabilities COMMAND crisp3ds capabilities)
  add_test(NAME cli_geometry COMMAND crisp3ds diagnose-geometry)
  add_test(NAME cli_unavailable COMMAND crisp3ds reconstruct ${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts/empty-project.json)
  set_tests_properties(cli_unavailable PROPERTIES WILL_FAIL TRUE)
  if(CRISP3DS_WITH_OPENCV)
    add_subdirectory(core/depth)
    add_library(crisp3ds_stereo_eval_lib STATIC core/evaluation/stereo_eval.cpp)
    target_include_directories(crisp3ds_stereo_eval_lib PUBLIC core/evaluation PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
    target_link_libraries(crisp3ds_stereo_eval_lib PUBLIC ${CRISP3DS_OPENCV_LIBRARIES})
    target_compile_features(crisp3ds_stereo_eval_lib PUBLIC cxx_std_20)
    add_executable(crisp3ds_stereo_eval core/evaluation/main.cpp)
    target_link_libraries(crisp3ds_stereo_eval PRIVATE crisp3ds_stereo_eval_lib)
    target_include_directories(crisp3ds_stereo_eval PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
    add_executable(crisp3ds_stereo_eval_tests core/evaluation/tests.cpp)
    target_link_libraries(crisp3ds_stereo_eval_tests PRIVATE crisp3ds_stereo_eval_lib)
    target_include_directories(crisp3ds_stereo_eval_tests PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
    add_test(NAME stereo_eval_synthetic COMMAND crisp3ds_stereo_eval_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME stereo_eval_prepared COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/core/evaluation/prepared_smoke.py
        $<TARGET_FILE:crisp3ds_stereo_eval> ${CMAKE_CURRENT_SOURCE_DIR}/.local-tools/tmp)
    endif()
    add_executable(crisp3ds_pose_integration core/tests/pose_integration.cpp)
    target_link_libraries(crisp3ds_pose_integration PRIVATE crisp3ds_core)
    target_include_directories(crisp3ds_pose_integration PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
    add_test(NAME pose_synthetic_image COMMAND crisp3ds_pose_integration
      ${CMAKE_CURRENT_BINARY_DIR}/synthetic-pose-fixture $<TARGET_FILE:crisp3ds>)
    add_executable(crisp3ds_sparse_integration core/tests/sparse_integration.cpp)
    target_link_libraries(crisp3ds_sparse_integration PRIVATE crisp3ds_core)
    target_include_directories(crisp3ds_sparse_integration PRIVATE ${CRISP3DS_OPENCV_INCLUDE_DIRS})
    add_test(NAME sparse_synthetic_image COMMAND crisp3ds_sparse_integration
      ${CMAKE_CURRENT_BINARY_DIR}/synthetic-sparse-fixture $<TARGET_FILE:crisp3ds>)
  endif()
endif()
